# 구현 계획 — ask_choice 번호 선택 입력 + vault 탐지

> **에이전트 실행자에게:** 태스크 단위로 진행한다. RED(실패 관찰) 없이 구현 코드를 쓰지 않는다.
> 태스크마다 자체 커밋으로 끝난다.

**목표:** 셸 스크립트에서 번호로 고를 수 있게 하고, `install/setup` 의 vault 경로
타이핑을 없앤다.

**전제(전 태스크 공통, 설계에서 확정):**
- **신규 에러코드 0개.** 탐지 실패는 에러가 아니라 열화다.
- **어떤 열화 경로도 설치를 막지 않는다.**
- `ask_choice` 의 stdout은 `$(...)`로 캡처된다 — **목록·안내는 전부 stderr로.**
  (`install/lib.sh:21-24` 의 `note()` 주석이 같은 함정을 이미 설명한다)
- 셸은 `set -euo pipefail` 환경에서 source된다. 기존 `say`/`warn`/`note` 헬퍼를 쓴다.
- **TTY 테스트는 `script -qec` 로 pty를 붙인다** (실측 확인: `[ -t 0 ]` 참, `read -rp` 가
  파이프 입력을 받음). pty 출력에는 `\r` 이 섞이므로 어서션 전에 `tr -d '\r'` 로 지운다.

# ---- 번호 선택 입력 (ask-choice) -------------------------------------------

# pty를 붙여 대화형 경로를 실행한다. 입력은 $1, 나머지는 bash 코드.
# 출력의 \r 을 제거해 반환한다.
run_with_tty() {
  local input="$1" code="$2"
  printf '%s' "$input" | script -qec "bash -c '$code'" /dev/null 2>&1 | tr -d '\r'
}

test_ask_choice() {
  group "번호 선택 입력 — ask_choice"

  local lib="${REPO_ROOT}/install/lib.sh"

  if ! grep -q '^ask_choice()' "$lib"; then
    fail_case "ask_choice 실존" "install/lib.sh 에 ask_choice 없음 (RED 기대 상태)"
    return
  fi

  local base="source ${lib}; ask_choice \"고르세요\" 3 \"가\" \"나\" \"다\""

  # (1) 유효 번호 선택
  local out
  out="$(run_with_tty '2\n' "$base")"
  assert_contains "$out" "2" "유효 번호 입력 → 그 번호 반환"

  # (2) 빈 입력 → 기본번호
  out="$(run_with_tty '\n' "echo \$($base)")"
  assert_contains "$out" "3" "빈 입력(Enter) → 기본번호 3 반환"

  # (3) 범위 밖 → 재질문 후 유효값 수용
  out="$(run_with_tty '9\n1\n' "echo RESULT=\$($base)")"
  assert_contains "$out" "RESULT=1" "범위 밖 입력 후 재질문 → 유효값 수용"
  assert_contains "$out" "1~3" "범위 밖 입력 시 허용 범위를 알려준다"

  # (4) 3회 연속 잘못된 입력 → 기본값 채택 (무한 루프 방지)
  out="$(run_with_tty 'x\ny\nz\n' "echo RESULT=\$($base)")"
  assert_contains "$out" "RESULT=3" "3회 실패 → 기본값 채택"

  # (5) 비TTY → 기본값 + 로그 (ask·ask_yn 과 동일 계약)
  out="$(bash -c "source ${lib}; ask_choice '고르세요' 2 '가' '나' '다'" </dev/null 2>&1)"
  assert_contains "$out" "2" "비대화형 → 기본번호 반환"
  assert_contains "$out" "비대화형" "비대화형 사실을 로그로 알린다"

  # (6) ★ 출력 오염 — stdout에는 오직 번호만 (목록이 섞이면 반환값이 깨진다)

### T2 — `ask_choice` (`install/lib.sh`)

- [ ] **2-1.** `ask_yn()` 정의 바로 뒤에 추가한다.

```bash
# ask_choice "질문 문구" "기본번호" "선택지1" "선택지2" ...
#   → 선택된 번호(1-based)를 stdout으로 echo
#
# 라벨이 아니라 번호를 반환한다 — 호출부가 자기 배열을 인덱싱하게 해서
# 표시 문자열과 값이 엉키지 않게 한다.
#
# ⚠ stdout은 $(...)로 캡처된다. 목록·안내는 전부 stderr로 나간다.
#   (위 note() 주석 참조 — 같은 함정이다)
ask_choice() {
  local prompt="$1" default="$2"
  shift 2
  local -a opts=("$@")
  local n=${#opts[@]} i ans tries=0

  if [ "$n" -eq 0 ]; then
    note "(선택지 없음) ${prompt} → 기본값 적용: ${default}"
    printf '%s\n' "$default"
    return 0
  fi

  if [ ! -t 0 ]; then
    note "(비대화형) ${prompt} → 기본 선택 적용: ${default}) ${opts[$((default - 1))]:-}"
    printf '%s\n' "$default"
    return 0
  fi

  while :; do
    printf '%s\n' "$prompt" >&2
    for ((i = 0; i < n; i++)); do
      if [ "$((i + 1))" = "$default" ]; then
        printf '  %d) %s   [기본]\n' "$((i + 1))" "${opts[$i]}" >&2
      else
        printf '  %d) %s\n' "$((i + 1))" "${opts[$i]}" >&2
      fi
    done
    read -rp "번호 입력 [${default}]: " ans
    ans="${ans:-$default}"

    if printf '%s' "$ans" | grep -qE '^[0-9]+$' && [ "$ans" -ge 1 ] && [ "$ans" -le "$n" ]; then
      printf '%s\n' "$ans"
      return 0
    fi

    tries=$((tries + 1))
    if [ "$tries" -ge 3 ]; then
      warn "입력을 3회 인식하지 못했습니다 — 기본값 ${default}을(를) 적용합니다."
      printf '%s\n' "$default"
      return 0
    fi
    warn "1~${n} 사이의 번호를 입력하세요 (입력값: '${ans}')"
  done
}
```

### T3 — `detect_obsidian_vaults` (`install/lib.sh`)

- [ ] **3-1.** `detect_superpowers()` 정의 뒤에 추가한다.

```bash
# detect_obsidian_vaults
#   → "<이름>\t<절대경로>" 를 한 줄씩 stdout. 탐지 실패 시 빈 출력(성공 종료).
#
# WSL/Windows 전용이다 — obsidian.json 이 Windows AppData 에 있다.
# Linux·macOS 는 표준 위치가 다르나 이 저장소에서 실측할 수 없어 다루지 않는다
# (설계 §4 비대상). 그 환경에서는 빈 출력 → 호출부가 자유 입력으로 열화한다.
#
# 탐지 실패를 에러로 만들지 않는다 — vault 지정은 선택이지 요건이 아니다.
detect_obsidian_vaults() {
  local cfg raw winpath unixpath

  cfg="$(ls /mnt/c/Users/*/AppData/Roaming/obsidian/obsidian.json 2>/dev/null | head -n1)"
  [ -n "$cfg" ] || return 0
  command -v python3 >/dev/null 2>&1 || return 0
  command -v wslpath >/dev/null 2>&1 || return 0

  raw="$(python3 - "$cfg" <<'PYEOF'
import json
import sys

try:
    with open(sys.argv[1], encoding="utf-8") as f:
        data = json.load(f)
except Exception:
    sys.exit(0)

vaults = data.get("vaults", {}) if isinstance(data, dict) else {}
if not isinstance(vaults, dict):
    sys.exit(0)

for v in vaults.values():
    if isinstance(v, dict) and v.get("path"):
        print(v["path"])
PYEOF
  )" || return 0
  [ -n "$raw" ] || return 0

  while IFS= read -r winpath; do
    [ -n "$winpath" ] || continue
    unixpath="$(wslpath -u "$winpath" 2>/dev/null)" || continue
    [ -d "$unixpath" ] || continue          # 삭제된 vault 가 목록에 남아 있을 수 있다
    printf '%s\t%s\n' "$(basename "$unixpath")" "$unixpath"
  done <<< "$raw"
}
```

승인된 설계 §4의 **7항목을 그대로 승계**한다. 편입된 항목 없음.

- **Linux·macOS vault 탐지를 하지 않는다.** 실측할 수 없는 것을 지원한다고 적지 않는다.
- **`install/update` 의 `ask_yn` 2곳(L85·91)을 바꾸지 않는다.** `y` 한 글자가 `2`+Enter보다 빠르다.
- **`install/setup` 의 설치 경로 입력(L56)·진행 확인(L75)을 바꾸지 않는다.**
- **탐지된 vault 가 Obsidian 에서 실제로 열리는지 확인하지 않는다.** 디렉토리 실존까지만 본다.
- **세션 계층의 질문 도구와 `ask_choice` 를 코드로 통합하지 않는다.** 규격만 같은 방향으로 맞춘다.
- **열린 쟁점 O1(문서 status 전이 시점)은 이번 범위 밖이다.**
- **`install/lib.sh` 의 기존 함수를 리팩터링하지 않는다.** 추가만 한다.
